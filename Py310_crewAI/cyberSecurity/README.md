## Code for SSH



```




import paramiko

VM_IP = "192.168.1.123"   # replace with your VM's IP
USERNAME = "your_username"
PASSWORD = "your_password"

# This is what your CrewAI agent produced
results = "ping -c 4 8.8.8.8"

ssh = paramiko.SSHClient()
ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())

ssh.connect(
    VM_IP,
    username=USERNAME,
    password=PASSWORD
)

stdin, stdout, stderr = ssh.exec_command(results)

print("COMMAND:")
print(results)

print("\nOUTPUT:")
print(stdout.read().decode())

print("ERRORS:")
print(stderr.read().decode())

ssh.close()

```

import paramiko
import re

# ============================================================
# VM SETTINGS
# ============================================================

VM_IP = "192.168.1.100"       # CHANGE
VM_PORT = 2222                # CHANGE
USERNAME = "your_username"    # CHANGE
PASSWORD = "your_password"    # CHANGE


# ============================================================
# CONNECT TO UBUNTU VM
# ============================================================

ssh = paramiko.SSHClient()
ssh.set_missing_host_key_policy(paramiko.AutoAddPolicy())

ssh.connect(
    VM_IP,
    port=VM_PORT,
    username=USERNAME,
    password=PASSWORD
)

print("Connected to VM.")


# ============================================================
# 1. EXECUTE A COMMAND GENERATED BY THE LLM
# ============================================================

# Replace this with your CrewAI result
results = "ping -c 4 8.8.8.8"

# Clean up LLM output
command = results.replace("\n", " ").strip()

print("\nCOMMAND:")
print(command)

stdin, stdout, stderr = ssh.exec_command(command)

output = stdout.read().decode()
error = stderr.read().decode()

print("\nOUTPUT:")
print(output)

if error:
    print("\nERROR:")
    print(error)


# ============================================================
# 2. READ AUTHENTICATION LOG
# ============================================================

# Ubuntu versions differ:
# /var/log/auth.log is common when rsyslog is being used.
# journalctl works on systemd systems.

log_command = "journalctl -u ssh --no-pager -n 50"

stdin, stdout, stderr = ssh.exec_command(log_command)

log_data = stdout.read().decode()
log_error = stderr.read().decode()

print("\n================ AUTH LOG ================\n")
print(log_data)

if log_error:
    print(log_error)


# ============================================================
# SEND log_data TO YOUR LLM HERE
# ============================================================

prompt = f"""
You are a cybersecurity analyst.

Analyze the following SSH authentication log.

Identify:
1. Failed login attempts
2. Successful logins
3. Suspicious IP addresses
4. Anything that might indicate an attack

LOG:

{log_data}
"""

print("\n================ PROMPT FOR LLM ================\n")
print(prompt)

# Example:
#
# analysis = your_crewai_function(prompt)
# print(analysis)


# ============================================================
# 3. EXAMPLE PYTHON SCRIPT THAT AN LLM COULD GENERATE
# ============================================================

generated_python_code = r'''
import subprocess
import re
from collections import Counter

result = subprocess.run(
    ["journalctl", "-u", "ssh", "--no-pager", "-n", "200"],
    capture_output=True,
    text=True
)

log = result.stdout

ips = re.findall(
    r"Failed password.*from ([0-9]+\.[0-9]+\.[0-9]+\.[0-9]+)",
    log
)

counts = Counter(ips)

print("FAILED SSH LOGIN REPORT")
print("-----------------------")

if not counts:
    print("No failed SSH logins found.")

for ip, count in counts.most_common():
    print(f"{ip}: {count} failed attempts")

    if count >= 3:
        print(f"  WARNING: {ip} may be suspicious")
'''


# ============================================================
# SEND THE PYTHON SCRIPT TO THE VM
# ============================================================

remote_file = "/tmp/analyze_ssh.py"

sftp = ssh.open_sftp()

with sftp.file(remote_file, "w") as f:
    f.write(generated_python_code)

sftp.close()

print("\nPython script uploaded to:")
print(remote_file)


# ============================================================
# RUN THE SCRIPT ON THE VM
# ============================================================

stdin, stdout, stderr = ssh.exec_command(
    f"python3 {remote_file}"
)

script_output = stdout.read().decode()
script_error = stderr.read().decode()

print("\n================ AGENT SCRIPT RESULT ================\n")
print(script_output)

if script_error:
    print("\nERROR:")
    print(script_error)


# ============================================================
# THIS OUTPUT COULD ALSO GO BACK TO THE LLM
# ============================================================

followup_prompt = f"""
You created a Python program to investigate failed SSH logins.

Here is the result from running the program on the Linux machine:

{script_output}

Explain what this result means and recommend the next diagnostic step.
"""

print("\n================ FOLLOW-UP FOR LLM ================\n")
print(followup_prompt)


# ============================================================
# CLOSE SSH
# ============================================================

ssh.close()

print("\nSSH connection closed.")
